package memory

import (
	"context"
	"errors"
	"sync"
	"time"

	"github.com/google/uuid"
	"github.com/LarssonOliver/inundated/internal/model"
)

type userRepositoryMem struct {
	mu    sync.RWMutex
	users map[uuid.UUID]*model.User
	subIndex map[string]uuid.UUID // map from sub to user ID for fast lookup
}

func NewUserRepository() *userRepositoryMem {
	return &userRepositoryMem{
		users: make(map[uuid.UUID]*model.User),
		subIndex: make(map[string]uuid.UUID),
	}
}

func (r *userRepositoryMem) GetByID(ctx context.Context, userID uuid.UUID) (*model.User, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	
	user, ok := r.users[userID]
	if !ok {
		return nil, errors.New("user not found")
	}
	return user, nil
}

func (r *userRepositoryMem) GetBySub(ctx context.Context, sub string) (*model.User, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	
	userID, ok := r.subIndex[sub]
	if !ok {
		return nil, errors.New("user not found")
	}
	return r.users[userID], nil
}

func (r *userRepositoryMem) Create(ctx context.Context, user *model.User) (*model.User, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	
	if _, ok := r.subIndex[user.Sub]; ok {
		return nil, errors.New("user with this sub already exists")
	}
	
	user.ID = uuid.New()
	user.CreatedAt = time.Now()
	user.UpdatedAt = time.Now()
	
	r.users[user.ID] = user
	r.subIndex[user.Sub] = user.ID
	
	return user, nil
}

func (r *userRepositoryMem) Update(ctx context.Context, user *model.User) (*model.User, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	
	if _, ok := r.users[user.ID]; !ok {
		return nil, errors.New("user not found")
	}
	
	user.UpdatedAt = time.Now()
	r.users[user.ID] = user
	
	return user, nil
}

package repository

import (
	"context"
	"fmt"
	"sync"
	"time"
	"github.com/google/uuid"
	"inundated/internal/model"
)

type MemoryUserRepository struct {
	users map[uuid.UUID]*model.User
	subToID map[string]uuid.UUID // Map OIDC sub claim to user ID
	mu    sync.RWMutex
}

func NewMemoryUserRepository() *MemoryUserRepository {
	return &MemoryUserRepository{
		users:   make(map[uuid.UUID]*model.User),
		subToID: make(map[string]uuid.UUID),
	}
}

func (r *MemoryUserRepository) GetByID(ctx context.Context, id uuid.UUID) (*model.User, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	
	user, ok := r.users[id]
	if !ok {
		return nil, fmt.Errorf("user not found")
	}
	return user, nil
}

func (r *MemoryUserRepository) GetBySub(ctx context.Context, sub string) (*model.User, error) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	
	id, ok := r.subToID[sub]
	if !ok {
		return nil, fmt.Errorf("user not found")
	}
	return r.users[id], nil
}

func (r *MemoryUserRepository) Create(ctx context.Context, user *model.User) error {
	r.mu.Lock()
	defer r.mu.Unlock()
	
	if _, ok := r.subToID[user.Sub]; ok {
		return fmt.Errorf("user with sub already exists")
	}
	
	now := time.Now()
	user.CreatedAt = now
	user.UpdatedAt = now
	
	r.users[user.ID] = user
	r.subToID[user.Sub] = user.ID
	return nil
}

func (r *MemoryUserRepository) Update(ctx context.Context, id uuid.UUID, update *model.UpdateUser) (*model.User, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	
	user, ok := r.users[id]
	if !ok {
		return nil, fmt.Errorf("user not found")
	}
	
	if update.Email != nil {
		user.Email = *update.Email
	}
	if update.Name != nil {
		user.Name = update.Name
	}
	user.UpdatedAt = time.Now()
	
	return user, nil
}

			token, err := jwt.Parse(tokenStr, func(token *jwt.Token) (interface{}, error) {
				// TODO: Validate signature against OIDC provider's JWKS endpoint
				return []byte(""), nil

			if err != nil {
				// For now, just parse claims without verification for development
				token, _, _ = new(jwt.Parser).ParseUnverified(tokenStr, jwt.MapClaims{})
			}
			
			claims, ok := token.Claims.(jwt.MapClaims)
			if !ok {
				return c.JSON(401, map[string]string{"error": "invalid token claims"})
			}
			
			// Extract OIDC standard claims
			sub, ok := claims["sub"].(string)
			if !ok || sub == "" {
				return c.JSON(401, map[string]string{"error": "missing sub claim"})
			}
			
			email, _ := claims["email"].(string)
			
			// Auto-provision user on first login
			user, err := userService.GetOrCreateBySub(c.Request().Context(), sub, email)
			if err != nil {
				return c.JSON(500, map[string]string{"error": "failed to provision user"})
			}
			
			// Add user context to request
			ctx := context.WithValue(c.Request().Context(), "user_id", user.ID)
			ctx = context.WithValue(ctx, "user_sub", user.Sub)
			c.SetRequest(c.Request().WithContext(ctx))
			
			return next(c)
		}
	}
}

	}
	
	return nil
